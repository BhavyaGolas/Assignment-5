def student_marks_lookup():
    # Step 1: Create dictionary of students and marks
    students = {
        "Alice": 85,
        "Bob": 92,
        "Charlie": 78,
        "David": 90
    }

    # Step 2: Ask user to input student name
    name = input("Enter the student's name: ")

    # Step 3 & 4: Retrieve marks or show error message
    if name in students:
        print(f"{name}'s marks: {students[name]}")
    else:
        print(f"Student '{name}' not found in records.")

# Run the function
student_marks_lookup()
